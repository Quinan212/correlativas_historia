\newcommand{\PageDecor}[1]{
  \begin{tikzpicture}[remember picture,overlay]
    \fill[APABg] (current page.south west) rectangle (current page.north east);
    \fill[APADeep] (current page.north west) rectangle ([yshift=-2.15cm]current page.north east);
    \fill[APAMid] ([yshift=-2.15cm]current page.north west) rectangle ([yshift=-2.95cm]current page.north east);
    \fill[APALight!88] ([yshift=-2.95cm]current page.north west) rectangle ([yshift=-3.22cm]current page.north east);
    \fill[APALight!18] ([xshift=1.05cm]current page.south west) rectangle ([xshift=1.32cm]current page.north west);
    \fill[APALight!22] ([xshift=-0.75cm,yshift=-5.4cm]current page.east) circle (1.45cm);
    \fill[APAMid!10] ([xshift=0.95cm,yshift=-8.2cm]current page.west) circle (0.92cm);
    \fill[white] ([xshift=-1.62cm,yshift=-1.06cm]current page.north east) circle (0.74cm);
    \node[font=\bfseries\Large,text=APADeep] at ([xshift=-1.62cm,yshift=-1.06cm]current page.north east) {#1};
    \fill[APADeep!8] (current page.south west) rectangle ([yshift=0.58cm]current page.south east);
  \end{tikzpicture}
  \vspace*{0.95cm}
}
